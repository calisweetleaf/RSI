cmake_minimum_required(VERSION 3.16)
project(recursive_weights VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(LAPACK REQUIRED)
add_library(rw src/c_api.cpp src/math.cpp src/weight.cpp src/runtime.cpp src/serialization.cpp src/pattern.cpp src/layer.cpp)
target_include_directories(rw PUBLIC include)
target_link_libraries(rw PUBLIC Threads::Threads OpenSSL::Crypto ${LAPACK_LIBRARIES})
add_executable(rw_cli src/main.cpp)
set_target_properties(rw_cli PROPERTIES OUTPUT_NAME rw)
target_link_libraries(rw_cli PRIVATE rw)
enable_testing()
add_executable(rw_tests tests/test_runtime.cpp)
target_link_libraries(rw_tests PRIVATE rw)
add_test(NAME runtime COMMAND rw_tests)
install(TARGETS rw rw_cli)
install(DIRECTORY include/ DESTINATION include)
